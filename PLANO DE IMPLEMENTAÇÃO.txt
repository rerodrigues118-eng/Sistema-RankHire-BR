Fim dos erros de mock: A página agora interage diretamente com as rotas de backend do Stripe (/api/stripe/checkout e /api/stripe/portal).

Gestão e Cancelamento Diretos: O botão de Portal do Cliente redireciona o usuário para a interface oficial do Stripe onde ele pode alterar cartões, baixar faturas ou cancelar a assinatura com um clique, sem intervenção manual.

            <div className="p-6 overflow-y-auto flex-1 bg-slate-50/30 space-y-6">
              {activeTab === 'overview' && (
                <div className="space-y-4">
                  <div className="p-4 bg-white rounded-xl border border-slate-200 shadow-xs space-y-3">
                    <div className="flex items-center justify-between text-xs">
                      <span className="text-slate-500 font-medium">Status</span>
                      <span className="text-slate-400">No status</span>
                    </div>
                    <div className="flex items-center justify-between text-xs">
                      <span className="text-slate-500 font-medium">Email</span>
                      <span className="text-slate-400">Checking email...</span>
                    </div>
                    <div className="flex items-center justify-between text-xs">
                      <span className="text-slate-500 font-medium">Phone</span>
                      <span className="text-slate-400">Checking number...</span>
                    </div>
                  </div>

                  <div className="space-y-3">
                    <h4 className="text-xs font-bold uppercase tracking-wider text-slate-400">Experience Summary</h4>
                    {selectedCandidate.history.map((h: any, i: number) => (
                      <div key={i} className="p-4 bg-white rounded-xl border border-slate-200 shadow-xs">

                    ))}

                </div>
              )}

              {activeTab === 'experience' && (

                  {selectedCandidate.history.map((h: any, i: number) => (
                    <div key={i} className="p-4 bg-white rounded-xl border border-slate-200 shadow-xs">

                    </div>
                  ))}
                </div>
              )}

              {activeTab === 'education' && (
                <div className="p-4 bg-white rounded-xl border border-slate-200 shadow-xs">
                  <h5 className="font-bold text-slate-900 text-sm">{selectedCandidate.education}</h5>
                  <p className="text-xs text-slate-500 mt-1">Jan 2009 - Jan 2014</p>
                </div>
              )}

              {activeTab === 'skills' && (
                <div className="flex flex-wrap gap-2">
                  {selectedCandidate.skills.map((s: string, i: number) => (

                      {s}
                    </span>
                  ))}
                </div>
              )}

              {activeTab === 'more' && (
                <div className="p-4 bg-white rounded-xl border border-slate-200 text-xs text-slate-600">
                  Additional notes, certifications, and portfolio links connected to this talent profile.
                </div>
              )}
            </div>

          </div>
        )}

      </div>
    </div>
  );
}
